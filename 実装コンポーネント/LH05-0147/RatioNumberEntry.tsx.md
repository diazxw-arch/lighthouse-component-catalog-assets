import {useId} from 'react';
import {cleanRatioValue} from './ratio-input';
import './learning-challenge.css';
type Props={heading:string;value:string;onValueChange:(value:string)=>void;onCheck:(value:string)=>void;onHint:()=>void;disabled?:boolean;busy?:boolean;error?:string;maxLength?:number};
export function RatioNumberEntry({heading,value,onValueChange,onCheck,onHint,disabled=false,busy=false,error,maxLength=15}:Props){const id=useId(),unavailable=disabled||busy,keys=['1','2','3','4','5','6','7','8','9','：','0','消す'],display=cleanRatioValue(value,maxLength),valid=/^\d+：\d+$/.test(display);const key=(label:string)=>{const next=label==='消す'?display.slice(0,-1):cleanRatioValue(display+label,maxLength);if(next!==display)onValueChange(next);};return <section className="lh-ratio-number" aria-labelledby={id+'-title'}><h2 id={id+'-title'}>{heading}</h2><svg className="lh-ratio-number__shapes" viewBox="0 0 88 112" role="img" aria-label="四角3個と三角3個">{[0,42,84].map(y=><g key={y}><rect x="0.5" y={y+.5} width="27" height="27" rx="3" fill="white" stroke="#1f6444"/><path d={`M72 ${y} 88 ${y+28}H56Z`} fill="#1f6444"/></g>)}</svg><input className="lh-ratio-number__input" aria-label="図形の数の比率" aria-invalid={!!error} aria-describedby={error?id+'-error':undefined} value={display} inputMode="text" autoComplete="off" disabled={unavailable} onChange={event=>onValueChange(cleanRatioValue(event.target.value,maxLength))} onKeyDown={event=>{if(event.key==='Enter'&&valid&&!unavailable){event.preventDefault();onCheck(display);}}}/><div className="lh-ratio-number__actions"><button type="button" aria-label="比率のヒントを見る" disabled={unavailable} onClick={onHint}>？</button><button type="button" disabled={unavailable||!valid} aria-busy={busy||undefined} onClick={()=>onCheck(display)}>{busy?'確認中…':'答えを確認する'}</button></div>{error&&<p id={id+'-error'} role="alert" className="lh-ratio-number__error">{error}</p>}<div className="lh-ratio-number__keypad" role="group" aria-label="比率の数字入力">{keys.map(label=><button key={label} type="button" data-key={label} disabled={unavailable} onClick={()=>key(label)}>{label}</button>)}</div></section>;}
