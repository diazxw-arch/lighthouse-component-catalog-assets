import {useRef} from 'react';
import './visual-search-selection.css';
export type ImageSearchResultType='images'|'places';
export interface ImageSearchResultTabsProps{value:ImageSearchResultType;onValueChange:(value:ImageSearchResultType)=>void;panelIds?:Record<ImageSearchResultType,string>;disabled?:boolean}
const options=[{id:'images',label:'関連する画像'},{id:'places',label:'場所を探す'}]as const;
export function ImageSearchResultTabs({value,onValueChange,panelIds,disabled=false}:ImageSearchResultTabsProps){const refs=useRef<(HTMLButtonElement|null)[]>([]);return <section className="lh-visual-search lh-image-result-tabs" aria-label="画像検索結果の種類"><span className="lh-image-result-tabs__handle" aria-hidden="true"/><div className="lh-image-result-tabs__list" role={panelIds?'tablist':'group'} aria-label="結果の種類">{options.map((option,index)=><button key={option.id} ref={el=>{refs.current[index]=el;}} type="button" className="lh-image-result-tabs__choice" role={panelIds?'tab':undefined} aria-selected={panelIds?value===option.id:undefined} aria-pressed={panelIds?undefined:value===option.id} aria-controls={panelIds?.[option.id]} tabIndex={panelIds?(value===option.id?0:-1):0} disabled={disabled} onClick={()=>onValueChange(option.id)} onKeyDown={event=>{if(disabled)return;let next;if(event.key==='ArrowLeft')next=(index+options.length-1)%options.length;else if(event.key==='ArrowRight')next=(index+1)%options.length;else if(event.key==='Home')next=0;else if(event.key==='End')next=options.length-1;else return;event.preventDefault();onValueChange(options[next].id);refs.current[next]?.focus();}}>{option.label}</button>)}</div></section>;}
