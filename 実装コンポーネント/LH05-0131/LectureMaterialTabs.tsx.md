import type {ReactNode} from 'react';
import * as Tabs from '@radix-ui/react-tabs';
import {PracticeGlyph} from './PracticeGlyph';
export type LectureMaterialTab='lectures'|'downloads'|'other';
export function LectureDownloadButton({label,onRequest,disabled=false,busy=false,tone='forest'}:{label:string;onRequest:()=>void;disabled?:boolean;busy?:boolean;tone?:'slate'|'forest'}){return <button type="button" className="lh-lecture-download" data-tone={tone} aria-label={label} aria-busy={busy||undefined} disabled={disabled||busy} onClick={onRequest}><span aria-hidden="true"><PracticeGlyph kind="download"/></span></button>;}
type Props={tab:LectureMaterialTab;onTabChange:(tab:LectureMaterialTab)=>void;onDownloadAll:()=>void;downloadAllLabel:string;downloadTone?:'slate'|'forest';downloadDisabled?:boolean;downloadBusy?:boolean;disabled?:boolean;children:ReactNode;panels:Record<'downloads'|'other',ReactNode>};
export function LectureMaterialTabs({tab,onTabChange,onDownloadAll,downloadAllLabel,downloadTone='slate',downloadDisabled=false,downloadBusy=false,disabled=false,children,panels}:Props){return <Tabs.Root className="lh-lecture-tabs" value={tab} onValueChange={value=>onTabChange(value as LectureMaterialTab)}><div className="lh-lecture-tabs__bar"><Tabs.List aria-label="講義と教材">{([{id:'lectures',label:'講義'},{id:'downloads',label:'保存した教材'},{id:'other',label:'その他'}] as const).map(item=><Tabs.Trigger key={item.id} value={item.id} disabled={disabled}>{item.label}</Tabs.Trigger>)}</Tabs.List><LectureDownloadButton label={downloadAllLabel} tone={downloadTone} disabled={disabled||downloadDisabled} busy={downloadBusy} onRequest={onDownloadAll}/></div><Tabs.Content value="lectures">{children}</Tabs.Content><Tabs.Content value="downloads">{panels.downloads}</Tabs.Content><Tabs.Content value="other">{panels.other}</Tabs.Content></Tabs.Root>;}
