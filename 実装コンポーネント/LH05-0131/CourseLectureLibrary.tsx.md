import type {ReactNode} from 'react';
import {LectureMaterialTabs,LectureDownloadButton,type LectureMaterialTab} from './LectureMaterialTabs';
import './lecture-dialogue.css';
export type CourseLecture={id:string;number:string;title:string;metadata:string};
export type LectureDownloadRequest={kind:'all'}|{kind:'chapter';id:string}|{kind:'lecture';id:string};
type Props={titleLines:readonly string[];instructor:string;chapterId:string;chapterLabel:string;lectures:readonly CourseLecture[];currentLectureId?:string;tab:LectureMaterialTab;onTabChange:(tab:LectureMaterialTab)=>void;panels:Record<'downloads'|'other',ReactNode>;onLecture:(id:string)=>void;onDownload:(request:LectureDownloadRequest)=>void;disabled?:boolean;busyDownloadId?:string;error?:string};
export function CourseLectureLibrary({titleLines,instructor,chapterId,chapterLabel,lectures,currentLectureId,tab,onTabChange,panels,onLecture,onDownload,disabled=false,busyDownloadId,error}:Props){return <section className="lh-course-lectures" aria-label="講義の一覧と教材の保存"><header><h2>{titleLines.map((line,index)=><span key={index}>{line}</span>)}</h2><p>{instructor}</p></header>{error&&<p role="alert" className="lh-lecture-message">{error}</p>}<LectureMaterialTabs tab={tab} onTabChange={onTabChange} panels={panels} disabled={disabled} downloadAllLabel="すべての教材を保存する" downloadDisabled={!lectures.length} downloadBusy={busyDownloadId==='all'} onDownloadAll={()=>onDownload({kind:'all'})}><div className="lh-course-lectures__chapter"><p>{chapterLabel}</p><LectureDownloadButton label="この章の教材を保存する" disabled={disabled||!lectures.length} busy={busyDownloadId==='chapter'} onRequest={()=>onDownload({kind:'chapter',id:chapterId})}/></div>{lectures.length?<ol>{lectures.map(lecture=><li key={lecture.id} data-current={lecture.id===currentLectureId}><button type="button" className="lh-course-lectures__open" disabled={disabled} aria-current={lecture.id===currentLectureId?'location':undefined} onClick={()=>onLecture(lecture.id)}><span className="lh-course-lectures__number">{lecture.number}</span><span className="lh-course-lectures__title">{lecture.title}</span><span className="lh-course-lectures__metadata">{lecture.metadata}</span></button><LectureDownloadButton label={`${lecture.title}の教材を保存する`} disabled={disabled} busy={busyDownloadId===lecture.id} onRequest={()=>onDownload({kind:'lecture',id:lecture.id})}/></li>)}</ol>:<p role="status" className="lh-lecture-message">講義はまだありません。</p>}</LectureMaterialTabs></section>;}
