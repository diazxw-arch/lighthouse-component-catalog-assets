import './syllabus-chapter-list.css';
export interface SyllabusChapter { id:string; chapter:string; title:string; disabled?:boolean; controls?:string }
export interface SyllabusChapterListProps { chapters:readonly SyllabusChapter[]; onOpen:(id:string)=>void; disabled?:boolean; busy?:boolean; error?:string; onRetry?:()=>void }
export function SyllabusChapterList({chapters,onOpen,disabled=false,busy=false,error,onRetry}:SyllabusChapterListProps){const unavailable=disabled||busy;return <section className="lh-syllabus-chapters" aria-label="シラバスの章"><ul aria-busy={busy||undefined}>{chapters.map(chapter=><li key={chapter.id}><button type="button" disabled={unavailable||chapter.disabled} aria-controls={chapter.controls} onClick={()=>{if(!unavailable&&!chapter.disabled)onOpen(chapter.id);}}><span className="lh-syllabus-chapters__chapter">{chapter.chapter}</span><span className="lh-syllabus-chapters__title">{chapter.title}</span><svg width="8" height="14" viewBox="0 0 8 14" aria-hidden="true"><path d="m1 1 6 6-6 6" fill="none" stroke="#526173" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/></svg></button></li>)}</ul>{!chapters.length&&!busy&&!error&&<p className="lh-syllabus-chapters__message">章はまだありません。</p>}{busy&&<p className="lh-syllabus-chapters__message" role="status">章を読み込んでいます…</p>}{error&&<p className="lh-syllabus-chapters__message" role="alert">{error}{onRetry&&<button type="button" disabled={unavailable} onClick={()=>{if(!unavailable)onRetry();}}>再試行</button>}</p>}</section>;}
