import {PracticeGlyph} from './PracticeGlyph';
import './vocabulary-practice.css';
import './first30-catalog-narrow.css';
type Props={headingLines:readonly string[];counts:readonly number[];learnedLabel:string;pages:readonly{id:string;label:string}[];currentPageId:string;onPage:(id:string)=>void;onClose:()=>void;disabled?:boolean};
export function VocabularyRetentionGuide({headingLines,counts,learnedLabel,pages,currentPageId,onPage,onClose,disabled=false}:Props){return <section className="lh-retention-guide" aria-label="覚えたことばの定着の仕組み"><button type="button" className="lh-retention-guide__close" aria-label="定着の説明を閉じる" onClick={onClose}><PracticeGlyph kind="close"/></button><h2>{headingLines.map((line,index)=><span key={index}>{line}</span>)}</h2><div className="lh-retention-guide__diagram"><svg className="lh-retention-guide__bridges" viewBox="0 -36 210 40" aria-hidden="true" fill="none" stroke="#1f6444" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round">{counts.slice(1).map((_,index)=><g key={index}><path d={`M${16+index*42} -8C${24+index*42} -31 ${50+index*42} -31 ${58+index*42} -8`}/><path d={`m${50+index*42} -12 8 4 2-8`}/></g>)}</svg><ol>{counts.map((count,index)=><li key={index}>{count}</li>)}</ol><p>{learnedLabel}</p></div><nav className="lh-retention-guide__pages" aria-label="定着の説明のページ">{pages.map(page=><button key={page.id} type="button" aria-label={page.label} aria-current={page.id===currentPageId?'step':undefined} disabled={disabled} onClick={()=>{if(page.id!==currentPageId)onPage(page.id);}}><span aria-hidden="true"/></button>)}</nav></section>;}
