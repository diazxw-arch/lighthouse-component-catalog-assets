import {useId} from 'react';
import {Switch} from '../ui/switch';
import {PracticeGlyph} from './PracticeGlyph';
import './learning-progress.css';
export type QuizSetupValue={count:string;answerMode:string;instant:boolean;trueFalse:boolean;multipleChoice:boolean;written:boolean};
export type QuizSetupOption={value:string;label:string};
type Props={heading:string;subject:string;counts:readonly QuizSetupOption[];answerModes:readonly QuizSetupOption[];value:QuizSetupValue;onChange:(value:QuizSetupValue)=>void;onStart:(value:QuizSetupValue)=>void;disabled?:boolean;busy?:boolean;error?:string};
export function QuizSetup({heading,subject,counts,answerModes,value,onChange,onStart,disabled=false,busy=false,error}:Props){const id=useId(),unavailable=disabled||busy,valid=counts.some(option=>option.value===value.count)&&answerModes.some(option=>option.value===value.answerMode)&&(value.trueFalse||value.multipleChoice||value.written);const toggle=(key:'instant'|'trueFalse'|'multipleChoice'|'written',label:string)=><div className={'lh-quiz-setup__'+key}><label htmlFor={id+'-'+key}>{label}</label><Switch id={id+'-'+key} type="button" checked={value[key]} disabled={unavailable} onCheckedChange={checked=>onChange({...value,[key]:checked})}/></div>;return <section className="lh-quiz-setup" aria-label="テストを設定"><h2>{heading}</h2><p className="lh-quiz-setup__subject">{subject}</p><div className="lh-quiz-setup__count"><label htmlFor={id+'-count'}>問題数</label><div className="lh-quiz-setup__select"><select id={id+'-count'} value={counts.some(option=>option.value===value.count)?value.count:''} disabled={unavailable||!counts.length} onChange={event=>onChange({...value,count:event.target.value})}>{!counts.some(option=>option.value===value.count)&&<option value="">選んでください</option>}{counts.map(option=><option key={option.value} value={option.value}>{option.label}</option>)}</select><span aria-hidden="true"><PracticeGlyph kind="chevron"/></span></div></div>{toggle('instant','答えたらすぐ判定する')}<div className="lh-quiz-setup__answer"><label htmlFor={id+'-answer'}>解答に使うもの</label><div className="lh-quiz-setup__select"><select id={id+'-answer'} value={answerModes.some(option=>option.value===value.answerMode)?value.answerMode:''} disabled={unavailable||!answerModes.length} onChange={event=>onChange({...value,answerMode:event.target.value})}>{!answerModes.some(option=>option.value===value.answerMode)&&<option value="">選んでください</option>}{answerModes.map(option=><option key={option.value} value={option.value}>{option.label}</option>)}</select><span aria-hidden="true"><PracticeGlyph kind="chevron"/></span></div></div>{toggle('trueFalse','正誤問題')}{toggle('multipleChoice','選択問題')}{toggle('written','記述問題')}<button type="button" className="lh-quiz-setup__start" disabled={unavailable||!valid} aria-busy={busy||undefined} onClick={()=>onStart({...value})}>{busy?'準備中…':'テストを始める'}</button>{error&&<p className="lh-quiz-setup__error" role="alert">{error}</p>}</section>;}
