import {useId} from 'react';
import './vocabulary-statistics.css';
export type ScoreDistributionData={headline:string;scoreLabel:string;youLabel:string;curvePath:string;markerX:number;markerY:number;averageX:number;axisLabel:string};
type Props={data:ScoreDistributionData|null;onPractice:()=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function LearningScoreDistribution({data,onPractice,disabled=false,busy=false,error,emptyLabel}:Props){const id=useId().replace(/:/g,''),clamp=(value:number,max:number)=>Number.isFinite(value)?Math.max(0,Math.min(max,value)):0,x=clamp(data?.markerX??0,318),y=clamp(data?.markerY??0,118),average=clamp(data?.averageX??0,318);return <section className="lh-score-distribution" aria-label="分布の中で学習の位置を知る"><article>{data?<><h3>{data.headline}</h3><p className="lh-score-distribution__score">{data.scoreLabel}</p><svg className="lh-score-distribution__chart" viewBox="0 0 318 118" preserveAspectRatio="none" role="img" aria-label={data.headline+'。'+data.scoreLabel+'。'+data.axisLabel}><defs><linearGradient id={id+'-marker'} x1="0" y1="0" x2="1" y2="1"><stop stopColor="#174c36"/><stop offset=".58" stopColor="#1f6444"/><stop offset="1" stopColor="#2d8f5e"/></linearGradient></defs><path d={data.curvePath} fill="none" stroke="#1f6444" strokeWidth="3" strokeLinecap="round"/><path d={`M0 118H318M${average} 0V118`} fill="none" stroke="#e4e4e4"/><path d={`M${x} ${y}V118`} fill="none" stroke={`url(#${id}-marker)`}/><circle cx={x+.5} cy={y+.5} r="6.5" fill="white" stroke={`url(#${id}-marker)`}/><text x={x+3} y={y-18} textAnchor="middle" fill="#1d1d1d" fontSize="12" fontWeight="700">{data.youLabel}</text></svg><p className="lh-score-distribution__axis">{data.axisLabel}</p></>:<p className="lh-score-distribution__empty" role="status">{emptyLabel}</p>}<button type="button" disabled={disabled||busy||!data} aria-busy={busy||undefined} onClick={onPractice}>{busy?'準備中…':'今すぐ練習する'}</button></article>{error&&<p className="lh-vocabulary-statistics-error" role="alert">{error}</p>}</section>;}
