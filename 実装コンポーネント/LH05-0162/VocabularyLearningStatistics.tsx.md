import {LearningMetricGrid,type LearningMetric} from './LearningMetricGrid';
import {LearningScoreDistribution,type ScoreDistributionData} from './LearningScoreDistribution';
import './vocabulary-statistics.css';
export type LearningWeekdayRecord={id:string;label:string;complete:boolean};
type Props={heading:string;testScore:string;testDateLabel:string;testDateTime?:string;streakHeading:string;streakValue:string;weekdays:readonly LearningWeekdayRecord[];metrics:readonly LearningMetric[];distribution:ScoreDistributionData|null;onPractice:()=>void;disabled?:boolean;busy?:boolean;error?:string;emptyMetricsLabel:string;emptyDistributionLabel:string};
export function VocabularyLearningStatistics({heading,testScore,testDateLabel,testDateTime,streakHeading,streakValue,weekdays,metrics,distribution,onPractice,disabled=false,busy=false,error,emptyMetricsLabel,emptyDistributionLabel}:Props){return <section className="lh-vocabulary-statistics" aria-label="単語学習の統計と現在の位置"><h2>{heading}</h2><article className="lh-vocabulary-statistics__test"><h3>{testScore}</h3><time dateTime={testDateTime}>{testDateLabel}</time></article><article className="lh-vocabulary-statistics__streak"><h3>{streakHeading}</h3><p>{streakValue}</p><ol aria-label="曜日ごとの学習記録">{weekdays.map(day=><li key={day.id}><span>{day.label}</span><span role="img" aria-label={day.label+'曜日：'+(day.complete?'学習済み':'記録なし')} data-complete={day.complete}/></li>)}</ol></article><LearningMetricGrid metrics={metrics} emptyLabel={emptyMetricsLabel}/><LearningScoreDistribution data={distribution} onPractice={onPractice} disabled={disabled} busy={busy} error={error} emptyLabel={emptyDistributionLabel}/></section>;}
