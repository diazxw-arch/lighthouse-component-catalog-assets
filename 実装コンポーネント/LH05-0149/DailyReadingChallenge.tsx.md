import {PracticeGlyph} from './PracticeGlyph';
import './learning-challenge.css';
export type ReadingChallengeDay={id:string;label:string;title:string;coverSrc:string;complete:boolean};
type Props={heading:string;subtitle:string;days:readonly ReadingChallengeDay[];currentDayId:string;trophyHeading:string;trophyDescription:string;trophySrc:string;onDay:(id:string)=>void;onRead:(id:string)=>void;onListen:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function DailyReadingChallenge({heading,subtitle,days,currentDayId,trophyHeading,trophyDescription,trophySrc,onDay,onRead,onListen,disabled=false,busy=false,error,emptyLabel}:Props){const current=days.find(day=>day.id===currentDayId),unavailable=disabled||busy;return <section className="lh-daily-reading" aria-label="一日一冊の読書チャレンジ"><h2>{heading}</h2><p className="lh-daily-reading__subtitle">{subtitle}</p><div className="lh-daily-reading__days" role="group" aria-label="チャレンジの日と本">{days.map(day=><article key={day.id}><button type="button" aria-label={day.label+(day.complete?'、完了':'、未完了')} aria-current={currentDayId===day.id?'step':undefined} disabled={unavailable} onClick={()=>{if(currentDayId!==day.id)onDay(day.id);}}>{day.complete?<PracticeGlyph kind="check"/>:<span className="lh-daily-reading__empty-check" aria-hidden="true"/>}{day.label}</button><img src={day.coverSrc} alt={day.title+'の表紙'}/></article>)}</div>{!days.length&&<p className="lh-daily-reading__empty" role="status">{emptyLabel}</p>}<div className="lh-daily-reading__trophies"><header><h3>{trophyHeading}</h3><p>{trophyDescription}</p></header><img src={trophySrc} alt="読書の達成を示すトロフィーの原画"/></div><div className="lh-daily-reading__actions"><button type="button" disabled={unavailable||!current} onClick={()=>{if(current)onRead(current.id);}}>読む</button><button type="button" disabled={unavailable||!current} aria-busy={busy||undefined} onClick={()=>{if(current)onListen(current.id);}}>聞く</button></div>{error&&<p className="lh-daily-reading__error" role="alert">{error}</p>}</section>;}
