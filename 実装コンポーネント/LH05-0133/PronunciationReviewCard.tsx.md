import {PracticeGlyph} from './PracticeGlyph';
import './lecture-dialogue.css';
type Props={phrase:string;meaning:string;completed:boolean;bookmarked:boolean;ownAudioAvailable:boolean;onFlag:()=>void;onBookmark:(value:boolean)=>void;onExampleAudio:()=>void;onOwnAudio:()=>void;onRestart:()=>void;examplePlaying?:boolean;ownPlaying?:boolean;disabled?:boolean;busy?:boolean;error?:string};
export function PronunciationReviewCard({phrase,meaning,completed,bookmarked,ownAudioAvailable,onFlag,onBookmark,onExampleAudio,onOwnAudio,onRestart,examplePlaying=false,ownPlaying=false,disabled=false,busy=false,error}:Props){const unavailable=disabled||busy||!phrase.trim();return <section className="lh-pronunciation-review" aria-label="発音のお手本と自分の声"><article><button type="button" className="lh-pronunciation-review__flag"
