import './reading-guide.css';
type Props={heading:string;bookId:string;title:string;author:string;coverSrc:string;actionLabel:string;onOpen:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function CurrentReadingBook({heading,bookId,title,author,coverSrc,actionLabel,onOpen,disabled=false,busy=false,error,emptyLabel}:Props){const empty=!bookId.trim();return <section className="lh-current-reading" aria-label="いま読んでいる本を開く"><h2>{heading}</h2>{empty?<p className="lh-current-reading__empty" role="status">{emptyLabel}</p>:<div className="lh-current-reading__book"><img src={coverSrc} alt={title+'の表紙'}/><div><h3>{title}</h3><p>{author}</p><button type="button" disabled={disabled||busy} aria-busy={busy||undefined} onClick={()=>onOpen(bookId)}>{busy?'開いています…':actionLabel}</button></div></div>}{error&&<p className="lh-current-reading__error" role="alert">{error}</p>}</section>;}
