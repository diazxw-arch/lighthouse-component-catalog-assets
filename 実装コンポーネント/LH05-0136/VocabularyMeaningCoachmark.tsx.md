import {useId,useLayoutEffect,useRef,useState} from 'react';
import {PracticeGlyph} from './PracticeGlyph';
import {EnergyInfinityGlyph,NewWordGlyph} from './VocabularyPracticeGlyph';
import './vocabulary-practice.css';
export type VocabularyChoice={id:string;label:string;row:1|2};
type Props={title:string;eyebrow:string;characterSrc:string;prefix:string;word:{id:string;text:string};suffix:string;tooltipLines:readonly string[];choices:readonly VocabularyChoice[];selectedIds:readonly string[];coachmarkOpen:boolean;progress:number;onDismissCoachmark:()=>void;onClose:()=>void;onEnergy:()=>void;onAudio:()=>void;onVocabulary:(id:string)=>void;onChoose:(id:string)=>void;onRemove:(id:string)=>void;onCheck:(ids:readonly string[])=>void;disabled?:boolean;busy?:boolean;error?:string};
export function VocabularyMeaningCoachmark({title,eyebrow,characterSrc,prefix,word,suffix,tooltipLines,choices,selectedIds,coachmarkOpen,progress,onDismissCoachmark,onClose,onEnergy,onAudio,onVocabulary,onChoose,onRemove,onCheck,disabled=false,busy=false,error}:Props){
 const id=useId().replace(/:/g,''),root=useRef<HTMLElement>(null),target=useRef<HTMLButtonElement>(null),tooltip=useRef<HTMLButtonElement>(null),wasOpen=useRef(false),[spot,setSpot]=useState({x:249,y:208,width:390});
 const unavailable=disabled||busy,selected=[...new Set(selectedIds)].map(selected=>choices.find(c=>c.id===selected)).filter((c):c is VocabularyChoice=>!!c);
 useLayoutEffect(()=>{let alive=true;const measure=()=>{const r=root.current?.getBoundingClientRect(),t=target.current?.getBoundingClientRect();if(alive&&r&&t&&r.width>0&&t.width>0)setSpot({x:t.x-r.x+t.width/2,y:t.y-r.y+t.height/2,width:r.width});};measure();const observer=typeof ResizeObserver==='undefined'?null:new ResizeObserver(measure);if(root.current)observer?.observe(root.current);if(target.current)observer?.observe(target.current);document.fonts?.ready.then(measure);return()=>{alive=false;observer?.disconnect();};},[word.text,prefix,suffix,coachmarkOpen]);
 useLayoutEffect(()=>{if(coachmarkOpen)tooltip.current?.focus();else if(wasOpen.current)target.current?.focus();wasOpen.current=coachmarkOpen;},[coachmarkOpen]);
 const tooltipLeft=Math.min(Math.max(16,spot.width-192),Math.max(16,spot.x-51)),amount=Number.isFinite(progress)?Math.max(0,Math.min(1,progress)):0;
 return <section ref={root} className="lh-meaning-coachmark" aria-label="ことばの意味を確認する練習案内" onKeyDown={event=>{if(coachmarkOpen&&event.key==='Escape'){event.preventDefault();event.stopPropagation();onDismissCoachmark();}}}><header><button type="button" className="lh-meaning-coachmark__close" aria-label="ことばの練習を閉じる" onClick={onClose}><PracticeGlyph kind="close"/></button><div role="progressbar" aria-label="練習の進み具合" aria-valuemin={0} aria-valuemax={100} aria-valuenow={Math.round(amount*100)} className="lh-meaning-coachmark__progress"><span style={{width:amount*100+'%'}}/></div><button type="button" className="lh-meaning-coachmark__energy" aria-label="学習のエネルギーを見る" disabled={unavailable||coachmarkOpen} onClick={onEnergy}><EnergyInfinityGlyph/></button></header><p className="lh-meaning-coachmark__eyebrow"><NewWordGlyph/>{eyebrow}</p><h2>{title}</h2><img className="lh-meaning-coachmark__character" src={characterSrc} alt="語句の練習を案内する人物"/><div className="lh-meaning-coachmark__sentence"><button type="button" className="lh-meaning-coachmark__audio" aria-label="文の音声を再生する" disabled={unavailable||coachmarkOpen} onClick={onAudio}><PracticeGlyph kind="speaker"/></button><p>{prefix}<button ref={target} type="button" className="lh-meaning-coachmark__word" disabled={unavailable} aria-label={'意味を開く：'+word.text} aria-describedby={coachmarkOpen?id+'-tip':undefined} onClick={()=>{onVocabulary(word.id);if(coachmarkOpen)onDismissCoachmark();}}>{word.text}</button>{suffix}</p></div><div className="lh-meaning-coachmark__answer" aria-label="組み立てた答え">{selected.map(choice=><button type="button" key={choice.id} disabled={unavailable||coachmarkOpen} aria-label={choice.label+'を答えから戻す'} onClick={()=>onRemove(choice.id)}>{choice.label}</button>)}</div><div className="lh-meaning-coachmark__choices">{([1,2] as const).map(row=><div key={row} data-row={row}>{choices.filter(c=>c.row===row).map(choice=><button type="button" key={choice.id} data-selected={selectedIds.includes(choice.id)} disabled={unavailable||coachmarkOpen||selectedIds.includes(choice.id)} onClick={()=>onChoose(choice.id)}>{choice.label}</button>)}</div>)}</div>{error&&<p className="lh-meaning-coachmark__error" role="alert">{error}</p>}<button type="button" className="lh-meaning-coachmark__check" disabled={unavailable||coachmarkOpen||!selected.length} aria-busy={busy||undefined} onClick={()=>onCheck(selected.map(c=>c.id))}>{busy?'確認中…':'確認する'}</button>
 {coachmarkOpen&&<><svg className="lh-meaning-coachmark__scrim" width="100%" height="100%" aria-hidden="true"><defs><mask id={id+'-spot'} maskUnits="userSpaceOnUse"><rect width="100%" height="100%" fill="white"/><circle cx={spot.x} cy={spot.y} r="32" fill="black"/></mask></defs><rect width="100%" height="100%" fill="#1d1d1d" opacity=".35" mask={`url(#${id}-spot)`}/></svg><div className="lh-meaning-coachmark__tooltip" role="dialog" aria-modal="false" aria-label="ことばの意味の案内" style={{left:tooltipLeft,top:spot.y+68}}><span className="lh-meaning-coachmark__pointer" style={{left:Math.max(8,Math.min(152,spot.x-tooltipLeft-8))}}/><button ref={tooltip} type="button" aria-label="ことばの案内を閉じる" aria-describedby={id+'-tip'} onClick={onDismissCoachmark}><span id={id+'-tip'}>{tooltipLines.map((line,index)=><span key={index}>{line}</span>)}</span></button></div></>}
 </section>;
}
