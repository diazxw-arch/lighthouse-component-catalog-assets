import {useCallback,useEffect,useRef,useState,type RefObject} from 'react';
export type QuantityKind='rice'|'water';
export function QuantityGlyph({kind}:{kind:QuantityKind}){return kind==='rice'?<svg viewBox="0 0 64 70" aria-hidden="true" fill="white" stroke="#1f6444" strokeWidth="3" strokeLinejoin="round"><path d="M8 14h48v50q0 4-4 4H12q-4 0-4-4Z"/><path d="M2 2h60l-6 12H8Z"/></svg>:<svg viewBox="0 0 82 58" aria-hidden="true" fill="white" stroke="#526173" strokeWidth="3" strokeLinejoin="round" strokeLinecap="round"><path d="M18 10H2v20M14 1h66l-8 55H27Z"/><path d="M21 27h55l-4 29H27Z" fill="#0bb3f0" stroke="none"/><path d="M14 1h66l-8 55H27Z" fill="none"/></svg>;}
type Drag={pointerId:number;element:HTMLButtonElement;startX:number;startY:number;x:number;y:number};
type Props={kind:QuantityKind;label:string;target:RefObject<HTMLDivElement>;onAdd:(kind:QuantityKind)=>void;disabled?:boolean};
export function QuantityToken({kind,label,target,onAdd,disabled=false}:Props){const active=useRef<Drag|null>(null),[offset,setOffset]=useState<{x:number;y:number}|null>(null);
 const clear=useCallback(()=>{const drag=active.current;active.current=null;setOffset(null);if(drag?.element.hasPointerCapture?.(drag.pointerId))drag.element.releasePointerCapture(drag.pointerId);},[]);
 useEffect(()=>{if(disabled)clear();},[disabled,clear]);useEffect(()=>()=>{const drag=active.current;active.current=null;if(drag?.element.hasPointerCapture?.(drag.pointerId))drag.element.releasePointerCapture(drag.pointerId);},[]);
 return <button className="lh-quantity-token" data-kind={kind} type="button" aria-label={label} disabled={disabled} style={offset?{transform:`translate(${offset.x}px,${offset.y}px)`,zIndex:4}:undefined} onPointerDown={event=>{if(disabled||event.button!==0||active.current)return;event.preventDefault();event.currentTarget.focus();try{event.currentTarget.setPointerCapture?.(event.pointerId);}catch{return;}active.current={pointerId:event.pointerId,element:event.currentTarget,startX:event.clientX,startY:event.clientY,x:event.clientX,y:event.clientY};setOffset({x:0,y:0});}} onPointerMove={event=>{const drag=active.current;if(drag?.pointerId!==event.pointerId)return;drag.x=event.clientX;drag.y=event.clientY;setOffset({x:drag.x-drag.startX,y:drag.y-drag.startY});}} onPointerUp={event=>{const drag=active.current;if(drag?.pointerId!==event.pointerId)return;const bounds=target.current?.getBoundingClientRect(),tap=Math.hypot(event.clientX-drag.startX,event.clientY-drag.startY)<6,inside=!!bounds&&event.clientX>=bounds.left&&event.clientX<=bounds.right&&event.clientY>=bounds.top&&event.clientY<=bounds.bottom;clear();if(tap||inside)onAdd(kind);}} onPointerCancel={event=>{if(active.current?.pointerId===event.pointerId)clear();}} onLostPointerCapture={event=>{if(active.current?.pointerId===event.pointerId)clear();}} onKeyDown={event=>{if(event.key==='Escape'&&active.current){event.preventDefault();event.stopPropagation();clear();}}} onBlur={clear} onClick={event=>{if(event.detail===0&&!active.current)onAdd(kind);}}><QuantityGlyph kind={kind}/></button>;
}
