import {useRef} from 'react';
import {QuantityGlyph,QuantityToken,type QuantityKind} from './QuantityToken';
import './learning-challenge.css';
type Props={heading:string;questionLines:readonly string[];characterSrc:string;rice:number;water:number;success:boolean;successLabel:string;onAdd:(kind:QuantityKind)=>void;onRemove:(kind:QuantityKind)=>void;onNext:()=>void;disabled?:boolean;busy?:boolean;error?:string};
export function QuantityRatioExercise({heading,questionLines,characterSrc,rice,water,success,successLabel,onAdd,onRemove,onNext,disabled=false,busy=false,error}:Props){const board=useRef<HTMLDivElement>(null),unavailable=disabled||busy,count=(value:number,max:number)=>Number.isFinite(value)?Math.max(0,Math.min(max,Math.floor(value))):0;return <section className="lh-quantity-ratio" aria-label="数量を動かして割合を考える"><h2>{heading}</h2><img className="lh-quantity-ratio__character" src={characterSrc} alt="ビーの原画"/><p className="lh-quantity-ratio__question">{questionLines.map((line,index)=><span key={index}>{line}</span>)}</p><div className="lh-quantity-ratio__divider"/><div ref={board} className="lh-quantity-ratio__board" role="group" aria-label="米と水の組合せ"><div className="lh-quantity-ratio__rice">{Array.from({length:count(rice,3)},(_,index)=><button type="button" key={index} aria-label={'米を1つ戻す：'+(index+1)} disabled={unavailable} onClick={()=>onRemove('rice')}><QuantityGlyph kind="rice"/></button>)}</div>{count(water,1)>0&&<button type="button" className="lh-quantity-ratio__water" aria-label="水を1つ戻す" disabled={unavailable} onClick={()=>onRemove('water')}><QuantityGlyph kind="water"/></button>}</div><div className="lh-quantity-ratio__tokens"><QuantityToken kind="rice" label="米を1つ加える。枠へ動かすか押してください" target={board} onAdd={onAdd} disabled={unavailable}/><QuantityToken kind="water" label="水を1つ加える。枠へ動かすか押してください" target={board} onAdd={onAdd} disabled={unavailable}/></div><div className="lh-quantity-ratio__result">{success&&<p role="status">{successLabel}</p>}{error&&<p role="alert">{error}</p>}<button type="button" disabled={unavailable||!success} aria-busy={busy||undefined} onClick={onNext}>次へ進む</button></div></section>;}
