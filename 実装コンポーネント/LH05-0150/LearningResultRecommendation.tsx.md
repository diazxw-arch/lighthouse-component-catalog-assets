import {PracticeGlyph} from './PracticeGlyph';
import './learning-challenge.css';
export type LearningRecommendation={id:string;title:string;author:string;metadata:string;coverSrc:string};
type Props={heading:string;results:readonly{id:string;label:string}[];recommendation:LearningRecommendation|null;recommendationHeading:string;reason:string;condition:string;nextLabel:string;canContinue:boolean;onListen:(id:string)=>void;onRead:(id:string)=>void;onContinue:()=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function LearningResultRecommendation({heading,results,recommendation,recommendationHeading,reason,condition,nextLabel,canContinue,onListen,onRead,onContinue,disabled=false,busy=false,error,emptyLabel}:Props){const unavailable=disabled||busy;return <section className="lh-learning-result" aria-label="学習結果から次の本へ進む"><h2>{heading}</h2><ul className="lh-learning-result__results">{results.map(result=><li key={result.id}><PracticeGlyph kind="check"/>{result.label}</li>)}</ul>{recommendation?<article className="lh-learning-result__recommendation"><h3>{recommendationHeading}</h3><p className="lh-learning-result__reason">{reason}</p><div className="lh-learning-result__book"><img src={recommendation.coverSrc} alt={recommendation.title+'の表紙'}/><div><h4>{recommendation.title}</h4><p>{recommendation.author}</p><p>{recommendation.metadata}</p></div></div><div className="lh-learning-result__actions"><button type="button" disabled={unavailable} onClick={()=>onListen(recommendation.id)}>聞く</button><button type="button" disabled={unavailable} onClick={()=>onRead(recommendation.id)}>読む</button></div></article>:<p className="lh-learning-result__empty" role="status">{emptyLabel}</p>}<p className="lh-learning-result__condition">{condition}</p><button type="button" className="lh-learning-result__continue" disabled={unavailable||!canContinue} aria-busy={busy||undefined} onClick={onContinue}>{nextLabel}</button>{error&&<p className="lh-learning-result__error" role="alert">{error}</p>}</section>;}
