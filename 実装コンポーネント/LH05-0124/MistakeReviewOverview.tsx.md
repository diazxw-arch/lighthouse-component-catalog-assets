import {ReaderChromeGlyph} from './ReaderChromeGlyph';
import './reader-utility.css';
export type MistakeReviewQuestion={id:string;label:string;reviewed?:boolean};
type Props={titleLines:readonly string[];actionLabel:string;questions:readonly MistakeReviewQuestion[];onBack:()=>void;onStart:()=>void;onQuestion:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string};
export function MistakeReviewOverview({titleLines,actionLabel,questions,onBack,onStart,onQuestion,disabled=false,busy=false,error}:Props){const unavailable=disabled||busy;return <section className="lh-mistake-review" aria-label="間違えた問題をまとめて復習" aria-busy={busy||undefined}><header><button className="lh-mistake-review__back" type="button" aria-label="学習に戻る" onClick={onBack}><ReaderChromeGlyph kind="back"/></button><h2>{titleLines.map((line,index)=><span key={index}>{line}</span>)}</h2><span className="lh-mistake-review__marker" aria-hidden="true"><ReaderChromeGlyph kind="repeat"/></span><button className="lh-mistake-review__start" type="button" disabled={unavailable||!questions.length} onClick={onStart}>{busy?'復習を準備しています…':actionLabel}</button></header>{questions.length?<><h3>{questions.length}つの問題</h3><ol>{questions.map(question=><li key={question.id}><button type="button" disabled={unavailable} onClick={()=>onQuestion(question.id)}><span>{question.label}</span><span className="lh-mistake-review__dot" role="img" data-reviewed={!!question.reviewed} aria-label={question.reviewed?'復習済み':'未復習'}/></button></li>)}</ol></>:<p role="status" className="lh-reader-message">復習する問題はまだありません。</p>}{error&&<p role="alert" className="lh-reader-message">{error}</p>}</section>;}
