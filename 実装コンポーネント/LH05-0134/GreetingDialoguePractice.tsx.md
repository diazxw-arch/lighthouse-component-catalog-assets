import {PracticeGlyph} from './PracticeGlyph';
import './lecture-dialogue.css';
export type GreetingPartnerMessage={id:string;text:string};
type Props={title:string;sectionLabel:string;description:string;questionLabel:string;firstPartner?:GreetingPartnerMessage;nextPartner?:GreetingPartnerMessage;prompt:string;response:{id:string;text:string;successful:boolean};hiddenMessageIds:readonly string[];onVisibility:(id:string,hidden:boolean)=>void;onClose:()=>void;onSettings:()=>void;onAudio:(id:string)=>void;onTranslate:(id:string)=>void;onRetry:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string};
type PartnerProps={message:GreetingPartnerMessage;first?:boolean;hidden:boolean;disabled:boolean;onAudio:(id:string)=>void;onTranslate:(id:string)=>void;onVisibility:(id:string,hidden:boolean)=>void};
function PartnerBubble({message,first=false,hidden,disabled,onAudio,onTranslate,onVisibility}:PartnerProps){return <div className="lh-greeting__partner" data-first={first}><p>{hidden?'発言を非表示にしました。':message.text}</p><div className="lh-greeting__utilities"><button type="button" className="lh-greeting__speaker" aria-label={`音声を再生する：${message.text}`} disabled={disabled} onClick={()=>onAudio(message.id)}><PracticeGlyph kind="speaker"/></button><button type="button" className="lh-greeting__translation" aria-label={`訳を開く：${message.text}`} disabled={disabled} onClick={()=>onTranslate(message.id)}>訳</button><button type="button" className="lh-greeting__visibility" aria-label={hidden?'発言を表示する':'発言を隠す'} aria-pressed={hidden} disabled={disabled} onClick={()=>onVisibility(message.id,!hidden)}><PracticeGlyph kind={hidden?'eye':'eye-off'}/></button></div></div>;}
export function GreetingDialoguePractice({title,sectionLabel,description,questionLabel,firstPartner,nextPartner,prompt,response,hiddenMessageIds,onVisibility,onClose,onSettings,onAudio,onTranslate,onRetry,disabled=false,busy=false,error}:Props){const unavailable=disabled||busy,partnerProps={disabled:unavailable,onAudio,onTranslate,onVisibility};return <section className="lh-greeting" aria-label="あいさつを練習する対話"><header><button type="button" className="lh-greeting__close" aria-label="対話の練習を閉じる" onClick={onClose}><PracticeGlyph kind="close"/></button><h2>{title}</h2><button type="button" className="lh-greeting__settings" aria-label="対話の設定を開く" disabled={disabled} onClick={onSettings}><PracticeGlyph kind="settings"/></button><p className="lh-greeting__section">{sectionLabel}</p><p className="lh-greeting__description">{description}</p><p className="lh-greeting__question"><span>{questionLabel}</span></p></header>{error&&<p role="alert" className="lh-lecture-message">{error}</p>}{firstPartner?<PartnerBubble {...partnerProps} first message={firstPartner} hidden={hiddenMessageIds.includes(firstPartner.id)}/>:<p role="status" className="lh-lecture-message">対話はまだありません。</p>}<div className="lh-greeting__prompt"><span aria-hidden="true"><PracticeGlyph kind="sparkles"/></span><p>{prompt}</p></div><div className="lh-greeting__response-row"><div className="lh-greeting__response">{response.successful&&<span className="lh-greeting__success" role="img" aria-label="呼び出し元が完了とした返答"><PracticeGlyph kind="check"/></span>}<p>{response.text||'返答はまだありません。'}</p><button type="button" className="lh-greeting__retry" aria-label="自分の返答をもう一度練習する" disabled={unavailable||!response.text.trim()} aria-busy={busy||undefined} onClick={()=>onRetry(response.id)}><PracticeGlyph kind="retry"/></button></div></div>{nextPartner&&<PartnerBubble {...partnerProps} message={nextPartner} hidden={hiddenMessageIds.includes(nextPartner.id)}/>}</section>;}
