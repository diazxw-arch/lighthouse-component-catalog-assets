import {useCallback,useEffect,useId,useRef,useState,type PointerEvent as ReactPointerEvent} from 'react';
import {VisualSearchIcon} from './VisualSearchIcons';
import {normalizePhotoRegion,resizePhotoRegion,photoRegionCornerTarget,type PhotoRegion,type PhotoRegionCorner} from './photo-region-geometry';
import './visual-search-selection.css';
export interface PhotoRegionPanelProps{photo:string;photoAlt:string;selection:PhotoRegion;onSelectionChange:(region:PhotoRegion)=>void;onSelectionCommit?:(region:PhotoRegion)=>void;onDone:(region:PhotoRegion)=>void;onCancel:()=>void;canUndo:boolean;onUndo:()=>void;canRedo:boolean;onRedo:()=>void;disabled?:boolean}
interface CornerDrag{pointerId:number;corner:PhotoRegionCorner;startX:number;startY:number;start:PhotoRegion;last:PhotoRegion;width:number;height:number;target:HTMLButtonElement;photo:string}
const corners=[{id:'nw',label:'左上',left:0,top:0},{id:'ne',label:'右上',left:100,top:0},{id:'sw',label:'左下',left:0,top:100},{id:'se',label:'右下',left:100,top:100}]as const;
export function PhotoRegionPanel({photo,photoAlt,selection,onSelectionChange,onSelectionCommit,onDone,onCancel,canUndo,onUndo,canRedo,onRedo,disabled=false}:PhotoRegionPanelProps){
 const description=useId(),bounds=description+'-bounds',media=useRef<HTMLDivElement>(null),drag=useRef<CornerDrag|null>(null),[size,setSize]=useState({width:310,height:552}),[minimumPhotoWidth,setMinimumPhotoWidth]=useState(88),[pointerActive,setPointerActive]=useState(false);
 const measure=useCallback(()=>{const box=media.current?.getBoundingClientRect();if(box&&box.width>=88&&box.height>=88)setSize(old=>Math.abs(old.width-box.width)<.1&&Math.abs(old.height-box.height)<.1?old:{width:box.width,height:box.height});},[]);
 useEffect(()=>{measure();const element=media.current;if(!element||typeof ResizeObserver==='undefined')return;const observer=new ResizeObserver(measure);observer.observe(element);return()=>observer.disconnect();},[measure]);
 useEffect(()=>{const d=drag.current;if(d&&(disabled||d.photo!==photo)){drag.current=null;setPointerActive(false);if(d.target.hasPointerCapture?.(d.pointerId))d.target.releasePointerCapture(d.pointerId);if(d.photo===photo)onSelectionChange(d.start);}},[disabled,photo,onSelectionChange]);
 const minimum={width:Math.max(.1,66/size.width),height:Math.max(.1,66/size.height)},region=normalizePhotoRegion(selection,minimum);
 function begin(corner:PhotoRegionCorner,event:ReactPointerEvent<HTMLButtonElement>){if(disabled||drag.current||event.button!==0)return;const box=media.current?.getBoundingClientRect();if(!box||box.width<88||box.height<88)return;event.preventDefault();setPointerActive(true);event.currentTarget.focus();event.currentTarget.setPointerCapture?.(event.pointerId);drag.current={pointerId:event.pointerId,corner,startX:event.clientX,startY:event.clientY,start:region,last:region,width:box.width,height:box.height,target:event.currentTarget,photo};}
 function cancelDrag(event?:ReactPointerEvent<HTMLButtonElement>){const d=drag.current;if(!d||event&&event.pointerId!==d.pointerId)return;drag.current=null;setPointerActive(false);if(d.target.hasPointerCapture?.(d.pointerId))d.target.releasePointerCapture(d.pointerId);onSelectionChange(d.start);}
 function move(event:ReactPointerEvent<HTMLButtonElement>){const d=drag.current;if(!d||d.pointerId!==event.pointerId)return;if(disabled){cancelDrag(event);return;}const next=resizePhotoRegion(d.start,d.corner,(event.clientX-d.startX)/d.width,(event.clientY-d.startY)/d.height,{width:Math.max(.1,66/d.width),height:Math.max(.1,66/d.height)});d.last=next;onSelectionChange(next);}
 function end(event:ReactPointerEvent<HTMLButtonElement>){const d=drag.current;if(!d||d.pointerId!==event.pointerId)return;if(disabled){cancelDrag(event);return;}drag.current=null;setPointerActive(false);if(d.target.hasPointerCapture?.(d.pointerId))d.target.releasePointerCapture(d.pointerId);if(JSON.stringify(d.last)!==JSON.stringify(d.start))onSelectionCommit?.(d.last);}
 return <section className="lh-visual-search lh-photo-region" aria-label="写真から範囲を選択"><div className="lh-photo-region__toolbar"><button type="button" aria-label="閉じる" disabled={disabled} onClick={onCancel}><VisualSearchIcon kind="close"/></button><button type="button" aria-label="元に戻す" disabled={disabled||!canUndo} onClick={onUndo}><VisualSearchIcon kind="undo"/></button><button type="button" aria-label="やり直す" disabled={disabled||!canRedo} onClick={onRedo}><VisualSearchIcon kind="redo"/></button><button type="button" className="lh-photo-region__done" disabled={disabled} onClick={()=>onDone(region)}>完了</button></div><div ref={media} className="lh-photo-region__media" style={{minWidth:minimumPhotoWidth}}><img src={photo} alt={photoAlt} onLoad={event=>{const image=event.currentTarget;if(image.naturalWidth>0&&image.naturalHeight>0)setMinimumPhotoWidth(Math.max(88,88*image.naturalWidth/image.naturalHeight));measure();}}/><div className="lh-photo-region__selection" style={{left:region.x*100+'%',top:region.y*100+'%',width:region.width*100+'%',height:region.height*100+'%'}}/><div className="lh-photo-region__controls">{corners.map(corner=>{const target=photoRegionCornerTarget(region,corner.id,size);return <button key={corner.id} type="button" className={'lh-photo-region__corner lh-photo-region__corner--'+corner.id} aria-label={corner.label+'の範囲を調整'} aria-describedby={description+' '+bounds} disabled={disabled} style={{left:target.left+22,top:target.top+22}} onPointerDown={event=>begin(corner.id,event)} onPointerMove={move} onPointerUp={end} onPointerCancel={cancelDrag} onLostPointerCapture={cancelDrag} onKeyDown={event=>{if(disabled||drag.current)return;let dx=0,dy=0;const step=event.shiftKey?.05:.01;if(event.key==='ArrowLeft')dx=-step;else if(event.key==='ArrowRight')dx=step;else if(event.key==='ArrowUp')dy=-step;else if(event.key==='ArrowDown')dy=step;else return;event.preventDefault();const next=resizePhotoRegion(region,corner.id,dx,dy,minimum);if(JSON.stringify(next)!==JSON.stringify(region)){onSelectionChange(next);onSelectionCommit?.(next);}}}><span aria-hidden="true" style={{transform:'translate('+(-target.offsetX)+'px,'+(-target.offsetY)+'px)'}}/></button>;})}</div></div><p id={description} className="lh-visual-search__sr">四隅をドラッグするか、矢印キーで範囲を調整できます。Shiftと矢印キーで大きく動かせます。</p><p id={bounds} className="lh-visual-search__sr" aria-live={pointerActive?'off':'polite'}>選択範囲: 左{Math.round(region.x*100)}%、上{Math.round(region.y*100)}%、幅{Math.round(region.width*100)}%、高さ{Math.round(region.height*100)}%</p></section>;
}
