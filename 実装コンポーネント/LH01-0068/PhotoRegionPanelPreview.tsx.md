import {useState} from 'react';import {createRoot} from 'react-dom/client';import {PhotoRegionPanel} from './PhotoRegionPanel';import {type PhotoRegion} from './photo-region-geometry';import photo from './assets/region-selection-original-photo.svg';import './discovery-preview.css';
const initial:PhotoRegion={x:12/310,y:12/512,width:286/310,height:488/512};
export function Preview(){const[history,setHistory]=useState([initial]),[index,setIndex]=useState(0),[selection,setSelection]=useState(initial),[message,setMessage]=useState('');function commit(region:PhotoRegion){setHistory(items=>[...items.slice(0,index+1),region]);setIndex(index+1);setSelection(region);}function restore(next:number){if(next<0||next>=history.length)return;setIndex(next);setSelection(history[next]);}return <main className="lh-discovery-preview lh-visual-search-preview--fullbleed"><div className="lh-visual-search-preview"><PhotoRegionPanel photo={photo} photoAlt="花を載せたトレーと手の写真" selection={selection} onSelectionChange={setSelection} onSelectionCommit={commit} canUndo={index>0} onUndo={()=>restore(index-1)} canRedo={index<history.length-1} onRedo={()=>restore(index+1)} onCancel={()=>{setSelection(history[index]);setMessage('閉じる操作を受け取りました。親画面は接続されていません。');}} onDone={()=>setMessage('選択範囲を受け取りました。画像検索サービスは接続されていません。')}/>{message&&<p role="status">{message}</p>}</div></main>;}createRoot(document.getElementById('root')!).render(<Preview/>);
