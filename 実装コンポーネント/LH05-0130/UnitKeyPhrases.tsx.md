import {GrammarVocabularyLine,type GrammarVocabularyPart} from './GrammarVocabulary';
import {GrammarAudioButton} from './GrammarAudioButton';
import './grammar-practice.css';
import './unit-keyphrases-ending.css';
export type UnitKeyPhrase={id:string;label:string;parts:readonly GrammarVocabularyPart[];meaning:string};
type Props={artworkSrc:string;metadata:string;titleLines:readonly string[];sectionLabel:string;headingLines:readonly string[];phrases:readonly UnitKeyPhrase[];onClose:()=>void;onVocabulary:(id:string)=>void;onAudio:(id:string,play:boolean)=>void;playingId?:string;busyId?:string;disabled?:boolean;error?:string};
function KeyPhraseVocabularyLine({parts,onRequest,disabled}:{parts:readonly GrammarVocabularyPart[];onRequest:(id:string)=>void;disabled:boolean}){const tail=parts[parts.length-1],token=parts[parts.length-2];
 const grouped=typeof tail==='string'&&typeof token==='object'&&/^[\u3001\u3002\uff01\uff1f!?\u300d\u300f\u3011\u3009\u300b\uff09]+$/.test(tail);
 if(!grouped)return <GrammarVocabularyLine parts={parts} onRequest={onRequest} disabled={disabled}/>;
 return <><GrammarVocabularyLine parts={parts.slice(0,-2)} onRequest={onRequest} disabled={disabled}/><span className="lh-unit-keyphrases__ending"><GrammarVocabularyLine parts={parts.slice(-2)} onRequest={onRequest} disabled={disabled}/></span></>;
}
export function UnitKeyPhrases({artworkSrc,metadata,titleLines,sectionLabel,headingLines,phrases,onClose,onVocabulary,onAudio,playingId,busyId,disabled=false,error}:Props){return <section className="lh-unit-keyphrases" aria-label="単元の重要表現を聞く"><header><button type="button" aria-label="重要な表現を閉じる" onClick={onClose}><svg viewBox="0 0 20 20" aria-hidden="true" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round"><path d="m3 3 14 14M17 3 3 17"/></svg></button><img src={artworkSrc} alt="たき火で休む学習の案内役"/><p>{metadata}</p><h2>{titleLines.map((line,index)=><span key={index}>{line}</span>)}</h2></header><div className="lh-unit-keyphrases__main"><p className="lh-unit-keyphrases__label">{sectionLabel}</p><h3>{headingLines.map((line,index)=><span key={index}>{line}</span>)}</h3>{error&&<p role="alert" className="lh-grammar-message">{error}</p>}{phrases.length?<ul>{phrases.map(phrase=><li key={phrase.id}><GrammarAudioButton label={phrase.label} playing={playingId===phrase.id} busy={busyId===phrase.id} disabled={disabled} onRequest={()=>onAudio(phrase.id,playingId!==phrase.id)}/><p className="lh-unit-keyphrases__phrase"><KeyPhraseVocabularyLine parts={phrase.parts} onRequest={onVocabulary} disabled={disabled}/></p><p className="lh-unit-keyphrases__meaning">{phrase.meaning}</p></li>)}</ul>:<p role="status" className="lh-grammar-message">重要な表現はまだありません。</p>}</div></section>;}
