import './grammar-practice.css';
type Props={heading:string;description:string;reviewLabel:string;advancedLabel:string;onReview:()=>void;onAdvanced:()=>void;disabled?:boolean;busyAction?:'review'|'advanced';error?:string};
/** Panel content for the host's anchored level popover. Placement/open state stays with the host. */
export function LessonLevelActionPanel({heading,description,reviewLabel,advancedLabel,onReview,onAdvanced,disabled=false,busyAction,error}:Props){const unavailable=disabled||!!busyAction;return <section className="lh-lesson-level" aria-label="復習と上級問題を選ぶ案内"><article><h2>{heading}</h2><p>{description}</p><button type="button" disabled={unavailable} aria-busy={busyAction==='review'||undefined} onClick={onReview}>{busyAction==='review'?'準備しています…':reviewLabel}</button><button type="button" className="lh-lesson-level__advanced" disabled={unavailable} aria-busy={busyAction==='advanced'||undefined} onClick={onAdvanced}>{busyAction==='advanced'?'準備しています…':advancedLabel}</button></article>{error&&<p role="alert" className="lh-grammar-message">{error}</p>}</section>;}
