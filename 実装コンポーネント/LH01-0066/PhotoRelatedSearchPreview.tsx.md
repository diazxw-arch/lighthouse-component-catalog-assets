import {useState} from 'react';import {createRoot} from 'react-dom/client';import {PhotoRelatedSearch} from './PhotoRelatedSearch';import photo from './assets/related-search-original-photo.svg';import './discovery-preview.css';
const terms=[{id:'natural',label:'自然な装い'},{id:'cardigan',label:'カーディガン'},{id:'quiet',label:'落ち着いた色'}];
export function Preview(){const[selected,setSelected]=useState<string|null>(null),[pinned,setPinned]=useState(false),[message,setMessage]=useState('');return <main className="lh-discovery-preview lh-visual-search-preview--source-photo"><div className="lh-visual-search-preview"><PhotoRelatedSearch sourceOverlay photo={photo} photoAlt="自然な装いの写真" terms={terms} selectedTermId={selected} onTermSelect={setSelected} onSearch={id=>setMessage((terms.find(t=>t.id===id)?.label??'画像')+'の検索操作を受け取りました。検索サービスは接続されていません。')} pinned={pinned} onPinnedChange={setPinned} onShare={()=>setMessage('共有操作を受け取りました。共有先は接続されていません。')}/>{message&&<p role="status">{message}</p>}</div></main>;}createRoot(document.getElementById('root')!).render(<Preview/>);
