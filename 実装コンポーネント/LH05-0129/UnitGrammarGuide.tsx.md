import {useId,type ReactNode} from 'react';
import {ReadingAnnotationGlyph} from './ReadingAnnotationGlyph';
import {GrammarVocabularyLine,type GrammarVocabularyPart,type GrammarVocabularyToken} from './GrammarVocabulary';
import {GrammarComparisonTable,type GrammarComparisonRow} from './GrammarComparisonTable';
import {GrammarAudioButton} from './GrammarAudioButton';
import './grammar-practice.css';
export type GrammarAudioExample={id:string;label:string;parts:readonly GrammarVocabularyPart[]};
type Props={sectionTitle:string;lessonLabel:string;artworkSrc:string;lessonIcon:ReactNode;heading:string;nounLabel:string;nounOpen:boolean;onNounOpenChange:(open:boolean)=>void;introLines:readonly(readonly GrammarVocabularyPart[])[];columns:readonly[GrammarVocabularyToken,GrammarVocabularyToken];rows:readonly GrammarComparisonRow[];explanationLines:readonly(readonly GrammarVocabularyPart[])[];examples:readonly GrammarAudioExample[];verbLabel:string;verbOpen:boolean;onVerbOpenChange:(open:boolean)=>void;verbContent?:ReactNode;onBack:()=>void;onVocabulary:(id:string)=>void;onAudio:(id:string,play:boolean)=>void;playingId?:string;busyId?:string;disabled?:boolean;error?:string};
function Chevron({open}:{open:boolean}){return <svg viewBox="0 0 20 20" aria-hidden="true" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round"><path d={open?'m3 13 7-7 7 7':'m3 7 7 7 7-7'}/></svg>;}
export function UnitGrammarGuide({sectionTitle,lessonLabel,artworkSrc,lessonIcon,heading,nounLabel,nounOpen,onNounOpenChange,introLines,columns,rows,explanationLines,examples,verbLabel,verbOpen,onVerbOpenChange,verbContent,onBack,onVocabulary,onAudio,playingId,busyId,disabled=false,error}:Props){
 const id=useId();return <section className="lh-unit-grammar" aria-label="単元の文法ガイド"><header><button type="button" aria-label="学習に戻る" onClick={onBack}><ReadingAnnotationGlyph kind="previous"/></button><h2>{sectionTitle}</h2><p>{lessonLabel}</p><img src={artworkSrc} alt="文法を案内するフクロウ"/></header>
 {error&&<p role="alert" className="lh-grammar-message">{error}</p>}<div className="lh-unit-grammar__main"><h3><span aria-hidden="true">{lessonIcon}</span>{heading}</h3><section className="lh-unit-grammar__noun"><h4><button type="button" disabled={disabled} aria-expanded={nounOpen} aria-controls={id+'-noun'} onClick={()=>onNounOpenChange(!nounOpen)}>{nounLabel}<Chevron open={nounOpen}/></button></h4><div id={id+'-noun'} hidden={!nounOpen}>
  <p className="lh-unit-grammar__intro">{introLines.map((line,index)=><span key={index}><GrammarVocabularyLine parts={line} onRequest={onVocabulary} disabled={disabled}/></span>)}</p>
  <div className="lh-unit-grammar__table"><GrammarComparisonTable label="名詞の性の比較" columns={columns} rows={rows} onVocabulary={onVocabulary} disabled={disabled}/></div>
  <p className="lh-unit-grammar__explanation">{explanationLines.map((line,index)=><span key={index}><GrammarVocabularyLine parts={line} onRequest={onVocabulary} disabled={disabled}/></span>)}</p>
  {examples.length?<ul className="lh-unit-grammar__examples">{examples.map(example=><li key={example.id}><GrammarAudioButton label={example.label} playing={playingId===example.id} busy={busyId===example.id} disabled={disabled} onRequest={()=>onAudio(example.id,playingId!==example.id)}/><p><GrammarVocabularyLine parts={example.parts} onRequest={onVocabulary} disabled={disabled}/></p></li>)}</ul>:<p role="status" className="lh-grammar-message">音声の例文はまだありません。</p>}
 </div></section></div><footer><h4><button type="button" disabled={disabled} aria-expanded={verbOpen} aria-controls={id+'-verbs'} onClick={()=>onVerbOpenChange(!verbOpen)}>{verbLabel}<Chevron open={verbOpen}/></button></h4><div id={id+'-verbs'} hidden={!verbOpen}>{verbContent}</div></footer></section>;
}
