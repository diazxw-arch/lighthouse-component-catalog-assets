import {PracticeGlyph} from './PracticeGlyph';
import {StudyToggleRow} from './StudyToggleRow';
import {AnswerDirectionCard,type AnswerDirectionValue} from './AnswerDirectionCard';
import {QuestionFormatCard,type QuestionFormatValue} from './QuestionFormatCard';
import './study-settings.css';
export type LearningSettingsValue={shuffle:boolean;speech:boolean;direction:AnswerDirectionValue;formats:QuestionFormatValue};
type Props={heading:string;value:LearningSettingsValue;gradingLabel:string;onChange:(value:LearningSettingsValue)=>void;onInvalidDirection:(trigger:HTMLButtonElement|null)=>void;onGrading:()=>void;onClose:()=>void;disabled?:boolean;busy?:boolean;error?:string};
export function LearningSettingsSheet({heading,value,gradingLabel,onChange,onInvalidDirection,onGrading,onClose,disabled=false,busy=false,error}:Props){const unavailable=disabled||busy;return <section className="lh-learning-settings" aria-label="暗記カードの学習方式の設定シート"><header><button type="button" aria-label="学習の設定を閉じる" onClick={onClose}><PracticeGlyph kind="close"/></button><h2>{heading}</h2></header><div className="lh-study-settings-card lh-learning-settings__basic"><h3>基本</h3><div className="lh-study-settings-card__rows"><StudyToggleRow label="用語の順番を変える" checked={value.shuffle} onChange={checked=>onChange({...value,shuffle:checked})} disabled={unavailable}/><StudyToggleRow label="文章を読み上げる" checked={value.speech} onChange={checked=>onChange({...value,speech:checked})} disabled={unavailable}/></div></div><AnswerDirectionCard heading="何を答えるか" value={value.direction} onChange={direction=>onChange({...value,direction})} onInvalid={onInvalidDirection} disabled={unavailable}/><QuestionFormatCard heading="問題の形式" value={value.formats} onChange={formats=>onChange({...value,formats})} disabled={unavailable}/><div className="lh-study-settings-card lh-learning-settings__grading"><h3>答えの判定方法</h3><button type="button" disabled={unavailable} onClick={onGrading}><span>{gradingLabel}</span><PracticeGlyph kind="chevron"/></button></div>{error&&<p role="alert" className="lh-study-settings-error">{error}</p>}</section>;}
