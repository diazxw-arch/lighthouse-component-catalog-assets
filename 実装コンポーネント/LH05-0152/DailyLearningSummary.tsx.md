import {PracticeGlyph} from './PracticeGlyph';
import './learning-progress.css';
export type DailyLearningTask={id:string;label:string;iconSrc:string;iconLabel:string;complete:boolean};
type Props={heading:string;tasks:readonly DailyLearningTask[];completionLabel:string;onConfirm:()=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function DailyLearningSummary({heading,tasks,completionLabel,onConfirm,disabled=false,busy=false,error,emptyLabel}:Props){const allComplete=tasks.length>0&&tasks.every(task=>task.complete);return <section className="lh-daily-summary" aria-label="一日の学びを終えた案内"><h2>{heading}</h2>{tasks.length?<ol>{tasks.map(task=><li key={task.id}><img src={task.iconSrc} alt={task.iconLabel}/><p>{task.label}</p><span className="lh-daily-summary__check" role="img" aria-label={task.complete?'完了：'+task.label:'未完了：'+task.label} data-complete={task.complete}>{task.complete&&<PracticeGlyph kind="check"/>}</span></li>)}</ol>:<p className="lh-daily-summary__empty" role="status">{emptyLabel}</p>}<div className="lh-daily-summary__completion-slot">{allComplete&&<p className="lh-daily-summary__completion" role="status">{completionLabel}</p>}</div><button type="button" className="lh-daily-summary__confirm" disabled={disabled||busy||!allComplete} aria-busy={busy||undefined} onClick={onConfirm}>確認しました</button>{error&&<p className="lh-daily-summary__error" role="alert">{error}</p>}</section>;}
