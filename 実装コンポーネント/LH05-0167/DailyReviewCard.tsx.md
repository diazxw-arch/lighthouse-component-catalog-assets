import './review-practice.css';
type Props={activityLabel:string;positionLabel:string;completed:number;total:number;courseId:string;courseTitle:string;heading:string;artSrc:string;artLabel:string;actionLabel:string;onReview:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function DailyReviewCard({activityLabel,positionLabel,completed,total,courseId,courseTitle,heading,artSrc,artLabel,actionLabel,onReview,disabled=false,busy=false,error,emptyLabel}:Props){const percentage=Number.isFinite(completed)&&Number.isFinite(total)&&total>0?Math.max(0,Math.min(100,completed/total*100)):0;return <section className="lh-daily-review" aria-label="今日の学びから復習に戻る"><header><h3>{activityLabel}</h3><p>{positionLabel}</p><div role="progressbar" aria-label={activityLabel} aria-valuemin={0} aria-valuemax={100} aria-valuenow={percentage}><span style={{width:percentage+'%'}}/></div></header><h2>{courseTitle}</h2><article>{courseId?<><h3>{heading}</h3><img src={artSrc} alt={artLabel}/></>:<p role="status">{emptyLabel}</p>}<button type="button" disabled={disabled||busy||!courseId} aria-busy={busy||undefined} onClick={()=>onReview(courseId)}>{busy?'準備中…':actionLabel}</button></article>{error&&<p className="lh-review-practice-error" role="alert">{error}</p>}</section>;}
