import './vocabulary-statistics.css';
export type MasteryReviewCourse={id:string;title:string;photoSrc:string;photoLabel:string;progress:number;progressLabel:string};
type Props={category:string;conceptLabel:string;mastery:number;masteryLabel:string;course:MasteryReviewCourse|null;scheduleLabel:string;onReview:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
const percent=(value:number)=>Number.isFinite(value)?Math.max(0,Math.min(100,value)):0;
export function CourseMasteryReview({category,conceptLabel,mastery,masteryLabel,course,scheduleLabel,onReview,disabled=false,busy=false,error,emptyLabel}:Props){return <section className="lh-course-mastery" aria-label="習得率と次の復習時期"><article><header><p>{category}</p><h3>{conceptLabel}</h3><h2>{masteryLabel}</h2><div className="lh-course-mastery__total" role="progressbar" aria-label="コース全体の習得率" aria-valuemin={0} aria-valuemax={100} aria-valuenow={percent(mastery)}><span style={{width:percent(mastery)+'%'}}/></div></header><div className="lh-course-mastery__course">{course?<><img src={course.photoSrc} alt={course.photoLabel}/><div><h3>{course.title}</h3><div className="lh-course-mastery__part" role="progressbar" aria-label={course.title+'の習得率'} aria-valuemin={0} aria-valuemax={100} aria-valuenow={percent(course.progress)}><span style={{width:percent(course.progress)+'%'}}/></div><p>{course.progressLabel}</p></div></>:<p role="status">{emptyLabel}</p>}</div><footer><p>{scheduleLabel}</p><button type="button" disabled={disabled||busy||!course} aria-busy={busy||undefined} onClick={()=>{if(course)onReview(course.id);}}>{busy?'準備中…':'復習を始める'}</button></footer>{error&&<p className="lh-vocabulary-statistics-error" role="alert">{error}</p>}</article></section>;}
