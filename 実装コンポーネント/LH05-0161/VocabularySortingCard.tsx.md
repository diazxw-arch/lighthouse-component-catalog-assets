import {useCallback,useEffect,useLayoutEffect,useId,useRef,useState} from 'react';
import {PracticeGlyph} from './PracticeGlyph';
import './vocabulary-statistics.css';
type Drag={pointerId:number;itemId:string;element:HTMLDivElement;startX:number};
type Props={positionLabel:string;learningCount:string;knownCount:string;statusLabel:string;itemId:string;word:string;instruction:string;canUndo:boolean;onReturnToLearning:(id:string)=>void;onUndo:()=>void;onNext:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function VocabularySortingCard({positionLabel,learningCount,knownCount,statusLabel,itemId,word,instruction,canUndo,onReturnToLearning,onUndo,onNext,disabled=false,busy=false,error,emptyLabel}:Props){const id=useId(),drag=useRef<Drag|null>(null),[offset,setOffset]=useState(0),unavailable=disabled||busy||!itemId||!word.trim();
 const release=useCallback(()=>{const current=drag.current;drag.current=null;if(current?.element.hasPointerCapture?.(current.pointerId))current.element.releasePointerCapture(current.pointerId);},[]);
 const clear=useCallback(()=>{release();setOffset(0);},[release]);useEffect(()=>()=>release(),[release]);useLayoutEffect(()=>{if(unavailable||(drag.current&&drag.current.itemId!==itemId))clear();},[unavailable,itemId,clear]);
 return <section className="lh-vocabulary-sort" aria-label="知っている語を横へ仕分けるカード"><p className="lh-vocabulary-sort__position">{positionLabel}</p><p className="lh-vocabulary-sort__learning" aria-label={'学習中の数：'+learningCount}>{learningCount}</p><p className="lh-vocabulary-sort__known" aria-label={'知っている数：'+knownCount}>{knownCount}</p><div className="lh-vocabulary-sort__back" aria-hidden="true"/><div className="lh-vocabulary-sort__front" role="group" aria-label={itemId?'単語：'+word:emptyLabel} aria-describedby={id+'-instruction'} aria-keyshortcuts="ArrowLeft Escape" tabIndex={unavailable?-1:0} style={offset?{transform:`translateX(${offset}px)`}:undefined}
 onPointerDown={event=>{if(unavailable||event.button!==0||drag.current)return;event.preventDefault();event.currentTarget.focus();try{event.currentTarget.setPointerCapture?.(event.pointerId);}catch{return;}drag.current={pointerId:event.pointerId,itemId,element:event.currentTarget,startX:event.clientX};}}
 onPointerMove={event=>{if(drag.current?.pointerId!==event.pointerId)return;setOffset(Math.max(-342,Math.min(120,event.clientX-drag.current.startX)));}}
 onPointerUp={event=>{const current=drag.current;if(current?.pointerId!==event.pointerId)return;const commit=!unavailable&&current.itemId===itemId&&event.clientX-current.startX<=-80;clear();if(commit)onReturnToLearning(current.itemId);}}
 onPointerCancel={event=>{if(drag.current?.pointerId===event.pointerId)clear();}} onLostPointerCapture={event=>{if(drag.current?.pointerId===event.pointerId)clear();}} onBlur={clear}
 onKeyDown={event=>{if(event.key==='Escape'&&drag.current){event.preventDefault();event.stopPropagation();clear();}if(event.key==='ArrowLeft'){event.preventDefault();if(!unavailable&&!event.repeat){clear();onReturnToLearning(itemId);}}}}>{itemId?<><h2>{statusLabel}</h2><p>{word}</p></>:<p className="lh-vocabulary-sort__empty" role="status">{emptyLabel}</p>}</div><button className="lh-vocabulary-sort__undo" type="button" aria-label="前の仕分けを取り消す" disabled={disabled||busy||!canUndo} onClick={onUndo}><PracticeGlyph kind="restart"/></button><p className="lh-vocabulary-sort__instruction" id={id+'-instruction'}>{instruction}</p><button className="lh-vocabulary-sort__next" type="button" aria-label="次の単語へ進む" aria-busy={busy||undefined} disabled={unavailable} onClick={()=>onNext(itemId)}><svg viewBox="0 0 14 18" aria-hidden="true"><path d="M0 0 14 9 0 18Z" fill="currentColor"/></svg></button>{error&&<p className="lh-vocabulary-statistics-error" role="alert">{error}</p>}</section>;
}
