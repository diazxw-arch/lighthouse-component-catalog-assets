import {PracticeGlyph} from './PracticeGlyph';
import {MicrophoneGlyph} from './VocabularyPracticeGlyph';
import {RecordingRequestButton} from './RecordingRequestButton';
import './vocabulary-practice.css';
import './first30-catalog-narrow.css';
type Props={letter:string;illustrationSrc:string;illustrationLabel:string;progress:number;recording:boolean;onClose:()=>void;onStart:()=>void;onStop:()=>void;onCancel:()=>void;disabled?:boolean;busy?:boolean;error?:string};
export function LetterSoundRecording({letter,illustrationSrc,illustrationLabel,progress,recording,onClose,onStart,onStop,onCancel,disabled=false,busy=false,error}:Props){const amount=Number.isFinite(progress)?Math.max(0,Math.min(1,progress)):0;return <section className="lh-letter-sound" aria-label="文字と音を結びつける練習"><button type="button" className="lh-letter-sound__close" aria-label="文字と音の練習を閉じる" onClick={onClose}><PracticeGlyph kind="close"/></button><div className="lh-letter-sound__progress" role="progressbar" aria-label="練習の進み具合" aria-valuemin={0} aria-valuemax={100} aria-valuenow={Math.round(amount*100)}><span style={{width:amount*100+'%'}}/></div><div className="lh-letter-sound__prompt"><img src={illustrationSrc} alt={illustrationLabel}/><h2>{letter}</h2></div><RecordingRequestButton mode="toggle" recording={recording} label={recording?'文字の録音を停止する':'文字の録音を始める'} className="lh-letter-sound__record" onStart={onStart} onStop={onStop} onCancel={onCancel} disabled={disabled||!letter.trim()} busy={busy}><span aria-hidden="true"/><MicrophoneGlyph/></RecordingRequestButton>{error&&<p className="lh-letter-sound__error" role="alert">{error}</p>}</section>;}
