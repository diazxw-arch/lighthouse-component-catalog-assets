import {useRef,useState} from 'react';
import {AnswerDirectionCard,type AnswerDirectionValue} from './AnswerDirectionCard';
import {MinimumAnswerNotice} from './MinimumAnswerNotice';
import './study-settings.css';
type Props={heading:string;cardHeading:string;value:AnswerDirectionValue;open:boolean;onChange:(value:AnswerDirectionValue)=>void;onOpenChange:(open:boolean)=>void;noticeTitle:string;noticeLines:readonly string[];noticeAction:string;disabled?:boolean;error?:string};
export function AnswerDirectionRequirement({heading,cardHeading,value,open,onChange,onOpenChange,noticeTitle,noticeLines,noticeAction,disabled=false,error}:Props){const[container,setContainer]=useState<HTMLElement|null>(null),returnFocus=useRef<HTMLButtonElement|null>(null);return <section ref={setContainer} className="lh-answer-requirement" aria-label="必須の出題条件を残す設定案内"><h2>{heading}</h2><AnswerDirectionCard heading={cardHeading} value={value} onChange={onChange} onInvalid={trigger=>{returnFocus.current=trigger;onOpenChange(true);}} disabled={disabled}/>{error&&<p className="lh-study-settings-error" role="alert">{error}</p>}<MinimumAnswerNotice open={open} container={container} returnFocus={returnFocus.current??container?.querySelector<HTMLButtonElement>('[role=switch]')??null} title={noticeTitle} descriptionLines={noticeLines} actionLabel={noticeAction} onDismiss={()=>onOpenChange(false)}/></section>;}
