import {useId} from 'react';
import {StudyToggleRow} from './StudyToggleRow';
import {PracticeGlyph} from './PracticeGlyph';
import './study-settings.css';
export type GradingChoice={id:string;label:string;descriptionLines:readonly string[]};
type Props={heading:string;choices:readonly GradingChoice[];selectedId:string;rewrite:boolean;canRewrite:boolean;rewriteDescriptionLines:readonly string[];onSelect:(id:string)=>void;onRewrite:(checked:boolean)=>void;onWritingMode:()=>void;disabled?:boolean;error?:string;emptyLabel:string};
export function AnswerGradingSettings({heading,choices,selectedId,rewrite,canRewrite,rewriteDescriptionLines,onSelect,onRewrite,onWritingMode,disabled=false,error,emptyLabel}:Props){const id=useId();return <section className="lh-answer-grading" aria-label="答えの厳密さを選ぶ採点設定"><div className="lh-study-settings-card"><h3 id={id+'-heading'}>{heading}</h3><fieldset aria-labelledby={id+'-heading'}>{choices.map(choice=><label key={choice.id} className="lh-answer-grading__choice"><span>{choice.label}</span><input type="radio" aria-label={choice.label} name={id} value={choice.id} checked={selectedId===choice.id} disabled={disabled} aria-describedby={id+'-'+choice.id+'-description'} onChange={()=>{if(!disabled&&choice.id!==selectedId)onSelect(choice.id);}}/><span id={id+'-'+choice.id+'-description'} className="lh-answer-grading__description">{choice.descriptionLines.map((line,index)=><span key={index}>{line}</span>)}</span></label>)}</fieldset>{!choices.length&&<p role="status">{emptyLabel}</p>}<StudyToggleRow label="正解を書き直す" checked={rewrite} onChange={onRewrite} disabled={disabled||!canRewrite}/></div><p className="lh-answer-grading__rewrite-description">{rewriteDescriptionLines.map((line,index)=><span key={index}>{line}</span>)}</p><button type="button" className="lh-answer-grading__writing-mode" disabled={disabled} onClick={onWritingMode}><span>記述モード</span><PracticeGlyph kind="chevron"/></button>{error&&<p className="lh-study-settings-error" role="alert">{error}</p>}</section>;}
