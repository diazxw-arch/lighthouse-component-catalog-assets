import {useState} from 'react';import {createRoot} from 'react-dom/client';import {AudiobookChapterList,type AudiobookTab} from './AudiobookChapterList';import {audiobookChapters} from './review-practice-demo-data';
import './review-practice-font.css';import './review-practice-preview.css';
export function AudiobookChapterListPreview(){const state=new URLSearchParams(location.search).get('state'),[tab,setTab]=useState<AudiobookTab>('chapters'),[message,setMessage]=useState('');return <main className="lh-review-practice-demo"><AudiobookChapterList tab={tab} onTabChange={setTab} chapters={state==='empty'?[]:audiobookChapters} panels={{about:<p className="lh-review-practice-panel">本についての情報はありません。</p>,related:<p className="lh-review-practice-panel">関連する本はありません。</p>}} onPlayAll={()=>setMessage('音声書籍を再生するリクエストです。')} onPlayChapter={id=>setMessage(id+'を再生するリクエストです。')} onChapterMenu={id=>setMessage(id+'の操作を開くリクエストです。')} disabled={state==='disabled'} busy={state==='busy'} error={state==='error'?'音声を準備できませんでした。':undefined} emptyLabel="章はまだありません。"/><p role="status">{message}</p></main>;}
createRoot(document.getElementById('root')!).render(<AudiobookChapterListPreview/>);
