import type {ReactNode} from 'react';
import * as Tabs from '@radix-ui/react-tabs';
import './review-practice.css';
export type AudiobookTab='about'|'chapters'|'related';
export type AudioChapter={id:string;title:string;durationLabel:string;disabled?:boolean};
type Props={tab:AudiobookTab;onTabChange:(tab:AudiobookTab)=>void;chapters:readonly AudioChapter[];onPlayAll:()=>void;onPlayChapter:(id:string)=>void;onChapterMenu:(id:string)=>void;playingId?:string;panels:Record<'about'|'related',ReactNode>;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function AudiobookChapterList({tab,onTabChange,chapters,onPlayAll,onPlayChapter,onChapterMenu,playingId,panels,disabled=false,busy=false,error,emptyLabel}:Props){return <section className="lh-audio-chapters" aria-label="音声書籍の章と長さの一覧"><Tabs.Root value={tab} onValueChange={value=>onTabChange(value as AudiobookTab)}><header><Tabs.List aria-label="音声書籍の情報">{([{id:'about',label:'本について'},{id:'chapters',label:'章の一覧'},{id:'related',label:'関連する本'}] as const).map(item=><Tabs.Trigger key={item.id} value={item.id} disabled={disabled}>{item.label}</Tabs.Trigger>)}</Tabs.List><button type="button" className="lh-audio-chapters__play" aria-label="音声書籍を再生する" aria-busy={busy||undefined} disabled={disabled||busy||!chapters.length} onClick={onPlayAll}><svg viewBox="0 0 14 20" aria-hidden="true"><path d="M0 0 14 10 0 20Z" fill="currentColor"/></svg></button></header><Tabs.Content value="chapters">{chapters.length?<ol>{chapters.map(chapter=><li key={chapter.id}><button className="lh-audio-chapters__chapter" type="button" aria-current={playingId===chapter.id?'true':undefined} disabled={disabled||busy||chapter.disabled} onClick={()=>onPlayChapter(chapter.id)}><span>{chapter.title}</span><span>{chapter.durationLabel}</span></button><button type="button" className="lh-audio-chapters__menu" aria-label={chapter.title+'の操作'} disabled={disabled||busy||chapter.disabled} onClick={()=>onChapterMenu(chapter.id)}>…</button></li>)}</ol>:<p className="lh-audio-chapters__empty" role="status">{emptyLabel}</p>}</Tabs.Content><Tabs.Content value="about">{panels.about}</Tabs.Content><Tabs.Content value="related">{panels.related}</Tabs.Content></Tabs.Root>{error&&<p className="lh-review-practice-error" role="alert">{error}</p>}</section>;}
