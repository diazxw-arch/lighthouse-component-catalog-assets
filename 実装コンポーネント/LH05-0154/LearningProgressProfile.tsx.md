import {LearningAchievementStrip,type LearningAchievement} from './LearningAchievementStrip';
import './learning-progress.css';
export type StudyWeekday={id:string;label:string};
type Props={heading:string;achievementHeading:string;achievements:readonly LearningAchievement[];achievementEmptyLabel:string;flameSrc:string;streakLabel:string;streakDescription:string;weekdays:readonly StudyWeekday[];selectedDayId:string;onDay:(id:string)=>void;onSuggest:()=>void;disabled?:boolean;busy?:boolean;error?:string};
export function LearningProgressProfile({heading,achievementHeading,achievements,achievementEmptyLabel,flameSrc,streakLabel,streakDescription,weekdays,selectedDayId,onDay,onSuggest,disabled=false,busy=false,error}:Props){return <section className="lh-progress-profile" aria-label="読書の達成と学習プロフィール"><h2>{heading}</h2><LearningAchievementStrip heading={achievementHeading} achievements={achievements} emptyLabel={achievementEmptyLabel}/><article className="lh-progress-profile__streak"><img src={flameSrc} alt="連続学習の炎の原画"/><h3>{streakLabel}</h3><p>{streakDescription}</p><div role="group" aria-label="学習の曜日">{weekdays.map(day=><button key={day.id} type="button" aria-label={day.label+'曜日'} aria-pressed={selectedDayId===day.id} disabled={disabled||busy} onClick={()=>{if(selectedDayId!==day.id)onDay(day.id);}}>{day.label}</button>)}</div></article><button type="button" className="lh-progress-profile__suggest" disabled={disabled||busy} aria-busy={busy||undefined} onClick={onSuggest}>さいころで次に読む要約を選ぶ</button>{error&&<p className="lh-progress-profile__error" role="alert">{error}</p>}</section>;}
