import {ReviewSpeakerGlyph} from './ReviewSpeakerGlyph';
import './review-practice.css';
export type UndoVocabulary={id:string;word:string;partOfSpeech:string;definition:string;example:string;progress:number;progressLabel:string};
export type VocabularyDeletion={id:string;message:string};
type Props={item:UndoVocabulary|null;deletion:VocabularyDeletion|null;onUndo:(id:string)=>void;onAudio:(id:string)=>void;onDelete:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string;emptyLabel:string};
export function VocabularyUndoCard({item,deletion,onUndo,onAudio,onDelete,disabled=false,busy=false,error,emptyLabel}:Props){const unavailable=disabled||busy,progress=item&&Number.isFinite(item.progress)?Math.max(0,Math.min(100,item.progress)):0;return <section className="lh-vocabulary-undo" aria-label="単語の説明を読み削除を取り消す">{deletion&&<div className="lh-vocabulary-undo__notice"><p role="status">{deletion.message}</p><button type="button" disabled={unavailable} onClick={()=>onUndo(deletion.id)}>元に戻す</button></div>}<article>{item?<><button className="lh-vocabulary-undo__audio" type="button" aria-label={item.word+'の音声を再生する'} disabled={unavailable} onClick={()=>onAudio(item.id)}><ReviewSpeakerGlyph/></button><h2>{item.word}</h2><p className="lh-vocabulary-undo__part">{item.partOfSpeech}</p><p className="lh-vocabulary-undo__definition">{item.definition}</p><p className="lh-vocabulary-undo__example">{item.example}</p><footer><div role="progressbar" aria-label={item.word+'の学習の進み具合'} aria-valuemin={0} aria-valuemax={100} aria-valuenow={progress}><span style={{width:progress+'%'}}/></div><p>{item.progressLabel}</p><button type="button" disabled={unavailable} aria-label={item.word+'を削除する'} onClick={()=>onDelete(item.id)}>削除</button></footer></>:<p className="lh-vocabulary-undo__empty" role="status">{emptyLabel}</p>}</article>{error&&<p className="lh-review-practice-error" role="alert">{error}</p>}</section>;}
