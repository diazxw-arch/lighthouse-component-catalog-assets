import {ReaderChromeGlyph,type ReaderChromeGlyphKind} from './ReaderChromeGlyph';
import './reader-utility.css';
export type ReadingBookTool='display'|'contents'|'highlights'|'notes'|'bookmarks';
type Props={title:string;status:string;onBack:()=>void;onDiscussion:()=>void;onTool:(tool:ReadingBookTool)=>void;disabled?:boolean};
const tools:readonly {id:ReadingBookTool;label:string;glyph?:ReaderChromeGlyphKind}[]=[{id:'display',label:'表示'},{id:'contents',label:'目次',glyph:'list'},{id:'highlights',label:'強調',glyph:'highlight'},{id:'notes',label:'ノート',glyph:'note'},{id:'bookmarks',label:'しおり',glyph:'tab'}];
export function ReadingBookUtilityHeader({title,status,onBack,onDiscussion,onTool,disabled=false}:Props){return <section className="lh-book-utility" aria-label="読書中の本と五つの操作"><header><button type="button" aria-label="本の一覧に戻る" onClick={onBack}><ReaderChromeGlyph kind="chevron-left"/></button><h2>{title}</h2><p>{status}</p><button type="button" aria-label="本の会話を開く" disabled={disabled} onClick={onDiscussion}><ReaderChromeGlyph kind="chat"/></button></header><nav aria-label="読書の操作">{tools.map(tool=><button type="button" key={tool.id} disabled={disabled} onClick={()=>onTool(tool.id)}><span className="lh-book-utility__circle" aria-hidden="true">{tool.glyph?<ReaderChromeGlyph kind={tool.glyph}/>:<span>あ</span>}</span><span>{tool.label}</span></button>)}</nav></section>;}
