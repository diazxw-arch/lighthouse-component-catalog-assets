import {useId} from 'react';
import './lecture-dialogue.css';
export type ListeningAnswer={id:string;label:string;disabled?:boolean};
type Props={progress:number;intro:string;instruction:string;sceneLines:readonly string[];avatarSrc:string;highlightedWord:string;phraseRemainder:string;meaning:string;answers:readonly ListeningAnswer[];selectedId?:string;correctId?:string;onAnswer:(id:string)=>void;disabled?:boolean;busy?:boolean;error?:string};
export function FinalListeningExercise({progress,intro,instruction,sceneLines,avatarSrc,highlightedWord,phraseRemainder,meaning,answers,selectedId,correctId,onAnswer,disabled=false,busy=false,error}:Props){const id=useId(),hasCorrectAnswer=!!selectedId&&selectedId===correctId&&answers.some(answer=>answer.id===selectedId),amount=Number.isFinite(progress)?Math.max(0,Math.min(1,progress)):0;return <section className="lh-final-listening" aria-label="場面を読んで聞く最後の問題"><div className="lh-final-listening__progress" role="progressbar" aria-label="問題の進み具合" aria-valuemin={0} aria-valuemax={100} aria-valuenow={Math.round(amount*100)}><span style={{width:`${amount*100}%`}}/></div><p className="lh-final-listening__intro">{intro}</p>{error&&<p role="alert" className="lh-lecture-message">{error}</p>}{busy&&<p role="status" className="lh-lecture-message">問題を確認しています…</p>}<p className="lh-final-listening__instruction">{instruction}</p><p className="lh-final-listening__scene">{sceneLines.map((line,index)=><span key={index}>{line}</span>)}</p><img className="lh-final-listening__avatar" src={avatarSrc} alt="会話に登場する女性"/><div className="lh-final-listening__bubble"><p><mark>{highlightedWord}</mark>{phraseRemainder}</p><p className="lh-final-listening__meaning">{meaning}</p></div><fieldset className="lh-final-listening__answers" aria-label={instruction} disabled={disabled||busy}>{answers.length?answers.map(answer=><label key={answer.id} data-selected={answer.id===selectedId} data-correct={answer.id===selectedId&&answer.id===correctId} data-disabled={!!answer.disabled}><input type="radio" name={id+'-answer'} value={answer.id} checked={answer.id===selectedId} aria-describedby={hasCorrectAnswer&&answer.id===selectedId?id+'-answer-result':undefined} disabled={answer.disabled||disabled||busy} onChange={()=>onAnswer(answer.id)}/><span>{answer.label}</span></label>):<p role="status">答えはまだありません。</p>}{hasCorrectAnswer&&<span id={id+'-answer-result'} className="lh-lecture-sr">正解として確認された答えです。</span>}</fieldset></section>;}
